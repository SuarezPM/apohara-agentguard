# --- kubectl read / inspect ---
kubectl get pods -A
kubectl get events --sort-by=.lastTimestamp
kubectl describe pod web-0
kubectl logs -f deploy/api
kubectl top nodes
kubectl config current-context
kubectl rollout status deploy/api

# --- kubectl mutating but non-destructive ---
kubectl apply -f k8s/
kubectl scale deployment api --replicas=3
kubectl set image deploy/api api=api:1.2.3
kubectl annotate pod web-0 team=platform
kubectl cordon node-1

helm search repo nginx
helm show values bitnami/nginx
helm install api ./charts/api
helm upgrade api ./charts/api --reuse-values
helm rollback api 2
helm template ./charts/api
helm lint ./charts/api
